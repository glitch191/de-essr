cmake_minimum_required(VERSION 3.22)

# The release workflow passes the version taken from the release tag.
set(DEESSR_VERSION "0.1.0" CACHE STRING "Plugin version (major.minor.patch)")

project(de-essr VERSION ${DEESSR_VERSION} LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Static MSVC runtime: the plugin does not depend on the Visual C++ redistributable.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

option(DEESSR_COPY_AFTER_BUILD "Copy the VST3 bundle to VST3_INSTALL_DIR after each build" ON)
set(VST3_INSTALL_DIR "C:/Program Files/Common Files/VST3" CACHE PATH
    "Folder the VST3 bundle is copied to when DEESSR_COPY_AFTER_BUILD is ON")
option(DEESSR_BUILD_PLUGIN "Build the plugin (OFF builds only the DSP core, its tests and the benchmark)" ON)
option(DEESSR_BUILD_TESTS "Build the offline DSP tests" ON)
option(DEESSR_BUILD_TOOLS "Build the benchmark and screenshot tools" ON)

include(FetchContent)

# Pinned dependencies: there is no lock file with CMake, the tags below are the lock.

# ---------------------------------------------------------------------------
# DSP core: plain C++, no dependency on JUCE, the host or the user interface.
# ---------------------------------------------------------------------------
add_library(deessr_dsp STATIC
    src/dsp/BandLayout.cpp
    src/dsp/BandLayout.h
    src/dsp/DeEsserEngine.cpp
    src/dsp/DeEsserEngine.h
    src/dsp/DspMath.h
    src/dsp/Fft.cpp
    src/dsp/Fft.h
    src/dsp/Params.h
    src/dsp/SibilanceDetector.cpp
    src/dsp/SibilanceDetector.h
    src/dsp/SpectralReducer.cpp
    src/dsp/SpectralReducer.h)
target_include_directories(deessr_dsp PUBLIC src)
if(MSVC)
    target_compile_options(deessr_dsp PRIVATE /W4 /permissive-)
endif()

# ---------------------------------------------------------------------------
# Plugin
# ---------------------------------------------------------------------------
if(DEESSR_BUILD_PLUGIN)

# JUCE 8.0.17-security = commit e548e97a54a083acbfc330d584419fd074409045
FetchContent_Declare(JUCE
    GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG 8.0.17-security
    GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)

juce_add_plugin(DeEssr
    PRODUCT_NAME "de-essr"
    COMPANY_NAME "de-essr"
    BUNDLE_ID "com.de-essr.de-essr"
    PLUGIN_MANUFACTURER_CODE Dssr
    PLUGIN_CODE Des1
    FORMATS VST3 Standalone
    IS_SYNTH FALSE
    NEEDS_MIDI_INPUT FALSE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS TRUE
    VST3_CATEGORIES Fx Dynamics)

# Local installation: after each Release build, copy the bundle to VST3_INSTALL_DIR.
# The script only writes this project's bundle and explains itself when the folder
# is not writable or when a DAW holds the file. The release workflow turns this off.
if(DEESSR_COPY_AFTER_BUILD AND WIN32)
    add_custom_command(TARGET DeEssr_VST3 POST_BUILD
        COMMAND powershell -NoProfile -ExecutionPolicy Bypass
                -File "${CMAKE_CURRENT_SOURCE_DIR}/scripts/install-vst3.ps1"
                -Source "$<GENEX_EVAL:$<TARGET_PROPERTY:DeEssr_VST3,JUCE_PLUGIN_ARTEFACT_FILE>>"
                -Destination "${VST3_INSTALL_DIR}"
                -Configuration "$<CONFIG>"
        COMMENT "Installing the VST3 bundle to ${VST3_INSTALL_DIR}"
        VERBATIM)
endif()

set(DEESSR_PLUGIN_SOURCES
    src/plugin/DisplayData.h
    src/plugin/Parameters.cpp
    src/plugin/Parameters.h
    src/plugin/PluginProcessor.cpp
    src/plugin/PluginProcessor.h
    src/ui/HistoryView.cpp
    src/ui/HistoryView.h
    src/ui/ParamControl.cpp
    src/ui/ParamControl.h
    src/ui/PluginEditor.cpp
    src/ui/PluginEditor.h
    src/ui/ReductionMeter.cpp
    src/ui/ReductionMeter.h
    src/ui/SpectrumView.cpp
    src/ui/SpectrumView.h
    src/ui/Theme.cpp
    src/ui/Theme.h
    src/ui/ToggleControls.cpp
    src/ui/ToggleControls.h)

# Definitions shared by the plugin and by the screenshot tool.
set(DEESSR_PLUGIN_DEFINITIONS
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_MODAL_LOOPS_PERMITTED=0
    DEESSR_PRODUCT_NAME="de-essr"
    DEESSR_VERSION_STRING="${PROJECT_VERSION}")

target_sources(DeEssr PRIVATE ${DEESSR_PLUGIN_SOURCES})
target_include_directories(DeEssr PUBLIC src)
target_compile_definitions(DeEssr PUBLIC ${DEESSR_PLUGIN_DEFINITIONS})

target_link_libraries(DeEssr
    PRIVATE
        deessr_dsp
        juce::juce_audio_utils
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags)

endif() # DEESSR_BUILD_PLUGIN

# ---------------------------------------------------------------------------
# Tests (DSP core only, synthetic signals generated by the test code)
# ---------------------------------------------------------------------------
if(DEESSR_BUILD_TESTS)
    FetchContent_Declare(Catch2
        GIT_REPOSITORY https://github.com/catchorg/Catch2.git
        GIT_TAG v3.9.1
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(Catch2)

    add_executable(DeEssrTests
        tests/TestSignals.cpp
        tests/TestSignals.h
        tests/EngineBasicsTests.cpp
        tests/DetectionTests.cpp
        tests/FftTests.cpp
        tests/RobustnessTests.cpp)
    target_link_libraries(DeEssrTests PRIVATE deessr_dsp Catch2::Catch2WithMain)
    if(MSVC)
        target_compile_options(DeEssrTests PRIVATE /W4 /permissive-)
    endif()

    enable_testing()
    list(APPEND CMAKE_MODULE_PATH "${catch2_SOURCE_DIR}/extras")
    include(Catch)
    catch_discover_tests(DeEssrTests)
endif()

# ---------------------------------------------------------------------------
# Tools (not shipped): CPU benchmark and editor screenshot renderer
# ---------------------------------------------------------------------------
if(DEESSR_BUILD_TOOLS)
    add_executable(DeEssrBench tools/bench/Main.cpp tests/TestSignals.cpp tests/TestSignals.h)
    target_include_directories(DeEssrBench PRIVATE tests)
    target_link_libraries(DeEssrBench PRIVATE deessr_dsp)

    if(DEESSR_BUILD_PLUGIN)
        # Compiles its own copy of the plugin sources, so it does not depend on
        # the plugin wrapper targets.
        juce_add_console_app(DeEssrScreenshot PRODUCT_NAME "de-essr-screenshot")
        target_sources(DeEssrScreenshot PRIVATE
            tools/screenshot/Main.cpp tests/TestSignals.cpp tests/TestSignals.h
            ${DEESSR_PLUGIN_SOURCES})
        target_include_directories(DeEssrScreenshot PRIVATE src tests)
        target_compile_definitions(DeEssrScreenshot PRIVATE ${DEESSR_PLUGIN_DEFINITIONS})
        target_link_libraries(DeEssrScreenshot
            PRIVATE
                deessr_dsp
                juce::juce_audio_utils
                juce::juce_recommended_config_flags
                juce::juce_recommended_warning_flags)

        # Loads the built VST3 the way a host does (see tools/hostcheck/Main.cpp).
        juce_add_console_app(DeEssrHostCheck PRODUCT_NAME "de-essr-hostcheck")
        target_sources(DeEssrHostCheck PRIVATE tools/hostcheck/Main.cpp)
        target_compile_definitions(DeEssrHostCheck PRIVATE
            JUCE_PLUGINHOST_VST3=1
            JUCE_WEB_BROWSER=0
            JUCE_USE_CURL=0
            JUCE_MODAL_LOOPS_PERMITTED=0)
        target_link_libraries(DeEssrHostCheck
            PRIVATE
                deessr_dsp
                juce::juce_audio_processors
                juce::juce_recommended_config_flags
                juce::juce_recommended_warning_flags)
    endif()
endif()
